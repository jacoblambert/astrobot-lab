cmake_minimum_required(VERSION 3.16)
project(glim_shadow_replay)
find_package(ament_cmake REQUIRED)
find_package(glim_ros REQUIRED)
find_package(glim REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(astrobot_interfaces REQUIRED)
find_package(image_transport REQUIRED)
add_executable(glim_shadow_replay replay.cpp)
target_compile_features(glim_shadow_replay PRIVATE cxx_std_17)
# Match the frozen installed libglim_ros ABI, verified by the preparation tool.
target_compile_definitions(glim_shadow_replay PRIVATE BUILD_WITH_CV_BRIDGE)
ament_target_dependencies(glim_shadow_replay glim_ros rosbag2_cpp rosgraph_msgs astrobot_interfaces image_transport)
target_link_libraries(glim_shadow_replay glim::glim)
# Opt-in candidate plugin links the installed IMU/smoother implementation. Rename
# CPU class symbols to avoid ELF interposition with the frozen libglim CPU class.
# Neither this target nor the probe is installed into the mission environment.
set(GLIM_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../../astrobot-lab/src/glim)
add_library(revision_target_cpu SHARED EXCLUDE_FROM_ALL
  ${GLIM_SOURCE}/src/glim/odometry/odometry_estimation_cpu.cpp
  ${GLIM_SOURCE}/src/glim/odometry/odometry_estimation_cpu_create.cpp)
target_include_directories(revision_target_cpu BEFORE PRIVATE ${GLIM_SOURCE}/include)
target_compile_features(revision_target_cpu PRIVATE cxx_std_17)
target_compile_options(revision_target_cpu PRIVATE -O2)
target_compile_definitions(revision_target_cpu PRIVATE
  OdometryEstimationCPU=RevisionTargetCPU OdometryEstimationCPUParams=RevisionTargetCPUParams)
target_link_libraries(revision_target_cpu glim::glim)
add_executable(test_revision_target EXCLUDE_FROM_ALL test_target.cpp)
target_include_directories(test_revision_target BEFORE PRIVATE ${GLIM_SOURCE}/include)
target_compile_features(test_revision_target PRIVATE cxx_std_17)
target_compile_definitions(test_revision_target PRIVATE
  OdometryEstimationCPU=RevisionTargetCPU OdometryEstimationCPUParams=RevisionTargetCPUParams)
target_link_libraries(test_revision_target revision_target_cpu glim::glim)
add_executable(test_absolute_schedule EXCLUDE_FROM_ALL test_absolute_schedule.cpp)
target_compile_features(test_absolute_schedule PRIVATE cxx_std_17)
ament_target_dependencies(test_absolute_schedule rosbag2_cpp geometry_msgs)
target_link_libraries(test_absolute_schedule glim::glim)
add_executable(glim_geometry_probe EXCLUDE_FROM_ALL geometry.cpp)
target_compile_features(glim_geometry_probe PRIVATE cxx_std_17)
target_compile_options(glim_geometry_probe PRIVATE -O2)
ament_target_dependencies(glim_geometry_probe rosbag2_cpp sensor_msgs)
target_link_libraries(glim_geometry_probe glim::glim)
add_executable(test_scan_geometry EXCLUDE_FROM_ALL test_scan_geometry.cpp)
target_include_directories(test_scan_geometry PRIVATE ${GLIM_SOURCE}/include)
target_compile_features(test_scan_geometry PRIVATE cxx_std_17)
target_link_libraries(test_scan_geometry Eigen3::Eigen)
set(GLIM_ROS_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../../astrobot-lab/src/glim_ros2)
add_library(closed_window_abs SHARED EXCLUDE_FROM_ALL ${GLIM_ROS_SOURCE}/src/glim_ros/abs_constraint_module.cpp)
target_include_directories(closed_window_abs BEFORE PRIVATE ${GLIM_ROS_SOURCE}/include)
target_compile_definitions(closed_window_abs PRIVATE AbsConstraintModule=ClosedWindowAbsConstraint)
target_compile_features(closed_window_abs PRIVATE cxx_std_17)
target_compile_options(closed_window_abs PRIVATE -O2)
ament_target_dependencies(closed_window_abs rclcpp geometry_msgs)
target_link_libraries(closed_window_abs glim::glim)
add_executable(test_closed_window EXCLUDE_FROM_ALL ${GLIM_ROS_SOURCE}/test/test_absolute_association.cpp)
target_include_directories(test_closed_window PRIVATE ${GLIM_ROS_SOURCE}/include)
target_compile_features(test_closed_window PRIVATE cxx_std_17)
